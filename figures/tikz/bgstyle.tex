% Shared style of the Background schematics (bg_*.tex).  Colors follow figures/bar_patterns.py: each hue keeps one
% meaning across the paper (orange = required output / WISEConv, slate steps = the baselines (tile skipping, gather-scatter
% extra data), dotted gray = wasted computation, as in fig:gap).  Drawn at \columnwidth (241 pt), included unscaled.  Cell fills and
% the dark-navy rules come from the authors' original schematics (input C1E5F5, weights F2CFEE, required FBE3D6);
% the SM fill is a lighter step of their D9D9D9 so that it stays apart from the wasted gray.

\usepackage[tt=false,type1=true]{libertine}
\usepackage[libertine]{newtxmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc,patterns.meta,positioning,fit,backgrounds,decorations.pathreplacing}

% \figframe{baseline|ours}{tag}: an outer frame around the whole schematic, placed last in the tikzpicture.  Baselines
% (fig:masked-conv, fig:tile-skip, fig:gather-scatter) get a dashed slate frame (the baselines' bar color), WISEConv
% (fig:overview) a solid orange one,
% each with its tag on the top edge.
\tikzset{
  baselineframe/.style={draw=baseline, line width=0.6pt, dash pattern=on 2.4pt off 1.4pt, rounded corners=3pt},
  oursframe/.style={draw=wiseconv!85!black, line width=0.9pt, rounded corners=3pt},
  baselinetag/.style={fill=white, inner xsep=1.4pt, inner ysep=0.6pt, anchor=west, text=baseline!80!black,
                      font=\fontsize{7.4}{8}\selectfont\bfseries},
  ourstag/.style={baselinetag, text=wiseconv!85!black},
}
\newcommand{\figframe}[2]{%
  \coordinate (ff-sw) at ($(current bounding box.south west)+(-0.9,-0.9)$);
  \coordinate (ff-ne) at ($(current bounding box.north east)+(0.9,2.0)$);
  \draw[#1frame] (ff-sw) rectangle (ff-ne);
  \node[#1tag] at ($(ff-sw|-ff-ne)+(3.0,0)$) {#2};}
